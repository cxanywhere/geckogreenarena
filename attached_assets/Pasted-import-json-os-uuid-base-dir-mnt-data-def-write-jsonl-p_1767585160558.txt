base_dir = "/mnt/data"

def write_jsonl(path, records):
    with open(path, "w", encoding="utf-8") as f:
        for r in records:
            f.write(json.dumps(r, ensure_ascii=False) + "\n")
    return path

def make_id(prefix):
    return f"{prefix}_{uuid.uuid4().hex[:12]}"
